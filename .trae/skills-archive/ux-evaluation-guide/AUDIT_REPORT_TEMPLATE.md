> # 当前（有 bug）：两个 App 都是蓝色
> # 期望（修复后）：音乐版输出 #00D8A6

# 期望（修复后）
curl -s http://127.0.0.1:{PORT}/src/style.css | grep "--accent:" | grep "{角色色值}"

# === §三 空状态检查 ===
grep "items.length.*===.*0\|!items\.length\|empty" {src/main.js 路径}

# === §五 错误可见性检查 ===

# === §三 Player 版默认 role 检查 ===

# === 部署后线上验证 ===
curl -s https://{域名}/manifest.webmanifest | grep theme_color
curl -s https://{域名}/src/style.css | grep "--accent:"
